# kenglish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a macOS menu-bar app that forces English input when a watched terminal/coding app becomes frontmost, with on/off toggle, settings, and launch-at-login.

**Architecture:** SwiftPM library `KenglishCore` (preferences, focus policy, input-source protocol) covered by unit tests; thin AppKit/SwiftUI app target `Kenglish` owns `MenuBarExtra`, workspace notifications, Carbon TIS, and `SMAppService`. No Accessibility permission in v1.

**Tech Stack:** Swift 5.9+, macOS 13+, SwiftUI `MenuBarExtra`, AppKit `NSWorkspace`, Carbon Text Input Source (TIS), `ServiceManagement.SMAppService`, XCTest, ad-hoc codesign.

## Global Constraints

- Bundle ID: `dev.finn.kenglish`
- Min OS: macOS 13.0
- English source order: `com.apple.keylayout.ABC` → `com.apple.keylayout.US` → first enabled keyboard layout with language `en`
- Leave unwatched apps alone (no auto-VI restore)
- Whole-app focus only (Cursor/VS Code entire window)
- No Accessibility permission in v1
- Work only in linked git worktree on non-protected branch

```swift
import Foundation
import Combine

public final class PreferencesStore: ObservableObject {
    private enum Key {
        static let enabled = "isEnabled"
        static let launchAtLogin = "launchAtLogin"
        static let watched = "watchedBundleIDs"
    }

    private let defaults: UserDefaults

    @Published public var isEnabled: Bool {
        didSet { defaults.set(isEnabled, forKey: Key.enabled) }
    }

    @Published public var launchAtLogin: Bool {
        didSet { defaults.set(launchAtLogin, forKey: Key.launchAtLogin) }
    }

    @Published public private(set) var watchedBundleIDs: [String] {
        didSet { defaults.set(watchedBundleIDs, forKey: Key.watched) }
    }

    public init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        if defaults.object(forKey: Key.enabled) == nil {
            self.isEnabled = true
        } else {
            self.isEnabled = defaults.bool(forKey: Key.enabled)
        }
        self.launchAtLogin = defaults.bool(forKey: Key.launchAtLogin)
        if let saved = defaults.array(forKey: Key.watched) as? [String] {
            self.watchedBundleIDs = WatchedApps.normalized(saved)
        } else {
            self.watchedBundleIDs = WatchedApps.defaults
        }
    }

    public func setWatchedBundleIDs(_ ids: [String]) {
        watchedBundleIDs = WatchedApps.normalized(ids)
    }

    public func addWatchedBundleID(_ id: String) {
        setWatchedBundleIDs(watchedBundleIDs + [id])
    }

    public func removeWatchedBundleID(_ id: String) {
        setWatchedBundleIDs(watchedBundleIDs.filter { $0 != id })
    }
}
```

```swift
public struct FocusDecision: Equatable {
    public let shouldSwitchToEnglish: Bool
    public let reason: String

    public init(shouldSwitchToEnglish: Bool, reason: String) {
        self.shouldSwitchToEnglish = shouldSwitchToEnglish
        self.reason = reason
    }
}

public enum FocusPolicy {
    public static func decide(
        isEnabled: Bool,
        frontBundleID: String?,
        watched: Set<String>,
        currentIsEnglish: Bool
    ) -> FocusDecision {
        guard isEnabled else {
            return FocusDecision(shouldSwitchToEnglish: false, reason: "disabled")
        }
        guard let id = frontBundleID, !id.isEmpty else {
            return FocusDecision(shouldSwitchToEnglish: false, reason: "no-front-app")
        }
        guard watched.contains(id) else {
            return FocusDecision(shouldSwitchToEnglish: false, reason: "unwatched")
        }
        if currentIsEnglish {
            return FocusDecision(shouldSwitchToEnglish: false, reason: "already-english")
        }
        return FocusDecision(shouldSwitchToEnglish: true, reason: "watched-needs-english")
    }
}
```

```swift
public enum EnglishSourceResolver {
    public static let abcID = "com.apple.keylayout.ABC"
    public static let usID = "com.apple.keylayout.US"

    public static func resolvePreferredID(
        availableIDs: [String],
        languagesByID: [String: [String]]
    ) -> String? {
        let set = Set(availableIDs)
        if set.contains(abcID) { return abcID }
        if set.contains(usID) { return usID }
        for id in availableIDs {
            let langs = languagesByID[id] ?? []
            if langs.contains(where: { $0.lowercased().hasPrefix("en") }) {
                return id
            }
        }
        return nil
    }

    public static func isEnglishID(
        _ id: String?,
        languagesByID: [String: [String]] = [:]
    ) -> Bool {
        guard let id else { return false }
        if id == abcID || id == usID { return true }
        let langs = languagesByID[id] ?? []
        return langs.contains(where: { $0.lowercased().hasPrefix("en") })
    }
}
```

- [ ] **Step 1: Implement** (manual API smoke only; no unit test of Carbon)

```swift
import Foundation
import Carbon
import KenglishCore

final class CarbonInputSourceSwitcher: InputSourceSwitching {
    func currentSourceID() -> String? {
        guard let src = TISCopyCurrentKeyboardInputSource()?.takeRetainedValue() else {
            return nil
        }
        return Self.stringProperty(src, key: kTISPropertyInputSourceID)
    }

    func englishSourceAvailable() -> Bool {
        preferredEnglishSource() != nil
    }

    func isEnglish(_ sourceID: String?) -> Bool {
        let map = languagesByID()
        return EnglishSourceResolver.isEnglishID(sourceID, languagesByID: map)
    }

    @discardableResult
    func selectEnglish() -> Bool {
        if isEnglish(currentSourceID()) { return true }
        guard let src = preferredEnglishSource() else { return false }
        let status = TISSelectInputSource(src)
        return status == noErr
    }

    private func preferredEnglishSource() -> TISInputSource? {
        let sources = enabledSources()
        let ids = sources.compactMap { Self.stringProperty($0, key: kTISPropertyInputSourceID) }
        let langs = languagesByID(from: sources)
        guard let preferred = EnglishSourceResolver.resolvePreferredID(
            availableIDs: ids,
            languagesByID: langs
        ) else { return nil }
        return sources.first {
            Self.stringProperty($0, key: kTISPropertyInputSourceID) == preferred
        }
    }

    private func enabledSources() -> [TISInputSource] {
        let keys = [kTISPropertyInputSourceIsEnabled: true] as CFDictionary
        guard let list = TISCreateInputSourceList(keys, false)?.takeRetainedValue() as? [TISInputSource] else {
            return []
        }
        return list.filter {
            Self.stringProperty($0, key: kTISPropertyInputSourceCategory) ==
                String(kTISCategoryKeyboardInputSource)
        }
    }

    private func languagesByID(from sources: [TISInputSource]? = nil) -> [String: [String]] {
        let srcs = sources ?? enabledSources()
        var map: [String: [String]] = [:]
        for src in srcs {
            guard let id = Self.stringProperty(src, key: kTISPropertyInputSourceID) else { continue }
            if let langs = Self.arrayProperty(src, key: kTISPropertyInputSourceLanguages) as? [String] {
                map[id] = langs
            }
        }
        return map
    }

    private static func stringProperty(_ src: TISInputSource, key: CFString) -> String? {
        guard let ptr = TISGetInputSourceProperty(src, key) else { return nil }
        return Unmanaged<CFString>.fromOpaque(ptr).takeUnretainedValue() as String
    }

    private static func arrayProperty(_ src: TISInputSource, key: CFString) -> NSArray? {
        guard let ptr = TISGetInputSourceProperty(src, key) else { return nil }
        return Unmanaged<CFArray>.fromOpaque(ptr).takeUnretainedValue() as NSArray
    }
}
```

```swift
import AppKit
import Combine
import KenglishCore

@MainActor
final class AppFocusMonitor: ObservableObject {
    @Published var lastWarning: String?

    private let preferences: PreferencesStore
    private let switcher: InputSourceSwitching
    private var observer: NSObjectProtocol?
    private var wakeObserver: NSObjectProtocol?
    private var debounceWork: DispatchWorkItem?
    private let debounceNs: UInt64 = 80_000_000 // 80ms

    init(preferences: PreferencesStore, switcher: InputSourceSwitching) {
        self.preferences = preferences
        self.switcher = switcher
    }

    func start() {
        stop()
        let center = NSWorkspace.shared.notificationCenter
        observer = center.addObserver(
            forName: NSWorkspace.didActivateApplicationNotification,
            object: nil,
            queue: .main
        ) { [weak self] _ in
            Task { @MainActor in self?.scheduleEvaluate() }
        }
        wakeObserver = center.addObserver(
            forName: NSWorkspace.didWakeNotification,
            object: nil,
            queue: .main
        ) { [weak self] _ in
            Task { @MainActor in self?.scheduleEvaluate() }
        }
        scheduleEvaluate()
    }

    func stop() {
        debounceWork?.cancel()
        if let observer {
            NSWorkspace.shared.notificationCenter.removeObserver(observer)
        }
        if let wakeObserver {
            NSWorkspace.shared.notificationCenter.removeObserver(wakeObserver)
        }
        observer = nil
        wakeObserver = nil
    }

    func scheduleEvaluate() {
        debounceWork?.cancel()
        let work = DispatchWorkItem { [weak self] in
            Task { @MainActor in self?.evaluateNow() }
        }
        debounceWork = work
        DispatchQueue.main.asyncAfter(deadline: .now() + .milliseconds(80), execute: work)
    }

    func evaluateNow() {
        guard preferences.isEnabled else { return }
        let frontID = NSWorkspace.shared.frontmostApplication?.bundleIdentifier
        let watched = Set(preferences.watchedBundleIDs)
        let currentID = switcher.currentSourceID()
        let isEN = switcher.isEnglish(currentID)
        let decision = FocusPolicy.decide(
            isEnabled: preferences.isEnabled,
            frontBundleID: frontID,
            watched: watched,
            currentIsEnglish: isEN
        )
        guard decision.shouldSwitchToEnglish else { return }
        if !switcher.englishSourceAvailable() {
            lastWarning = "No English keyboard found"
            return
        }
        lastWarning = nil
        _ = switcher.selectEnglish()
    }

    deinit {
        // observers removed on stop; MainActor deinit cannot call stop safely in all Swift versions
    }
}
```

```swift
enum LaunchAtLogin {

        SMAppService.mainApp.status == .enabled

    @discardableResult

        do {
            if enabled {
                try SMAppService.mainApp.register()
```
